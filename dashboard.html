Quick Reference

Copy-paste commands to get things done fast

# Install Warp (GPU sim framework)
$ pip install warp-lang

# Install 3D tools
$ pip install pxr trimesh Pillow numpy

# Clone PhysX 5 (open-source physics)
$ git clone --depth 1 https://github.com/NVIDIA-Omniverse/PhysX.git

# Run built-in Warp particle example
$ python -m warp.examples.example_particles

# Run built-in Warp cloth example
$ python -m warp.examples.example_cloth

# Run built-in Warp rigid body example
$ python -m warp.examples.example_rigid_body

# Check if CUDA is available for Warp
$ python -c "import warp as wp; wp.init(); print(wp.is_cuda_available())"

# Check GPU status
$ nvidia-smi
