/* ═════════════════════════════════════════════════════════════════════
   GAME 5 — NEON SPACE SHOOTER

const ssCanvas = document.getElementById('shooterCanvas');
const ssCtx = ssCanvas.getContext('2d');
const ssW = 480, ssH = 600;

const ssScoreEl = document.getElementById('ssScore');
const ssLivesEl = document.getElementById('ssLives');

let ssBest = 0;
try { ssBest = parseInt(window.localStorage.getItem('shooterBest'), 10) || 0; } catch (e) { ssBest = 0; }

const ss = {
    running: true, over: false,
    score: 0, lives: 3,
    frame: 0, invuln: 0, shake: 0, fireCd: 0, spawnTimer: 55,
    player: { x: ssW / 2, y: 500, w: 30, h: 34, speed: 4.6 },
    bullets: [], enemyBullets: [], enemies: [], particles: [], rings: [], stars: [],
    keys: { left: false, right: false, up: false, down: false, fire: false },
    firing: false,
    mouse: { active: false, x: 0, y: 0 },
};

const ssDroneColor = '#00f0ff';
const ssZigColor = '#ff4df0';
const ssDiverColor = '#ffb020';

if (!CanvasRenderingContext2D.prototype.roundRect) {
    CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
        if (r > w / 2) r = w / 2;
        if (r > h / 2) r = h / 2;
        this.moveTo(x + r, y);
        this.lineTo(x + w - r, y);
        this.quadraticCurveTo(x + w, y, x + w, y + r);
        this.lineTo(x + w, y + h - r);
        this.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
        this.lineTo(x + r, y + h);
        this.quadraticCurveTo(x, y + h, x, y + h - r);
        this.lineTo(x, y + r);
        this.quadraticCurveTo(x, y, x + r, y);
        return this;
    };
}

function ssSpawnStars() {
    ss.stars = [];
    for (let layer = 0; layer < 3; layer++) {
        const count = layer === 0 ? 46 : layer === 1 ? 28 : 16;
        for (let i = 0; i < count; i++) {
            ss.stars.push({
                x: Math.random() * ssW,
                y: Math.random() * ssH,
                size: 0.6 + layer * 0.55 + Math.random() * 0.6,
                speed: 0.25 + layer * 0.45 + Math.random() * 0.3,
                layer: layer,
                tw: Math.random() * Math.PI * 2,
            });
        }
    }
}

function ssPickType() {
    const f = ss.frame;
    const r = Math.random();
    if (f < 700) return 'drone';
    if (f < 1800) return r < 0.75 ? 'drone' : 'zigzag';
    if (f < 3000) return r < 0.5 ? 'drone' : (r < 0.85 ? 'zigzag' : 'diver');
    return r < 0.38 ? 'drone' : (r < 0.75 ? 'zigzag' : 'diver');
}

function ssEnemyFireTimer(e) {
    let lo = 90, hi = 170;
    if (e.type === 'zigzag') { lo = 120; hi = 210; }
    if (e.type === 'diver') { lo = 150; hi = 240; }
    const d = Math.min(60, Math.floor(ss.frame / 350));
    return Math.max(34, lo + Math.floor(Math.random() * (hi - lo)) - d);
}

function ssSpawnEnemy(type) {
    const t = type || ssPickType();
    const margin = 26;
    const e = {
        x: margin + Math.random() * (ssW - margin * 2),
        y: -30,
        r: 13,
        type: t,
        phase: Math.random() * Math.PI * 2,
        fireTimer: 60,
        dive: 0,
    };
    if (t === 'drone') {
        e.speed = 1.15 + Math.min(1.6, ss.frame / 9000);
        e.color = ssDroneColor;
        e.points = 10;
    } else if (t === 'zigzag') {
        e.speed = 0.95 + Math.min(1.3, ss.frame / 11000);
        e.color = ssZigColor;
        e.points = 20;
    } else {
        e.speed = 2.4 + Math.min(2.0, ss.frame / 9000);
        e.color = ssDiverColor;
        e.points = 30;
    }
    e.fireTimer = ssEnemyFireTimer(e);
    ss.enemies.push(e);
}

function ssUpdateSpawner() {
    ss.spawnTimer--;
    if (ss.spawnTimer > 0) return;
    ss.spawnTimer = Math.max(24, 78 - Math.floor(ss.frame / 420));
    let n = 1;
    if (ss.frame > 1200 && Math.random() < 0.22) n = 2 + Math.floor(Math.random() * 2);
    for (let i = 0; i < n; i++) ssSpawnEnemy();
}

function ssSpawnParticles(x, y, count, color, speed, life) {
    for (let i = 0; i < count; i++) {
        const ang = Math.random() * Math.PI * 2;
        const sp = 0.5 + Math.random() * speed;
        ss.particles.push({
            x: x, y: y,
            vx: Math.cos(ang) * sp,
            vy: Math.sin(ang) * sp,
            gravity: 0.04,
            life: Math.floor(life * (0.5 + Math.random() * 0.5)),
            maxLife: life,
            color: color,
            size: 1.5 + Math.random() * 3,
        });
    }
}

function ssFire() {
    if (ss.fireCd > 0) return;
    ss.fireCd = 9;
    const p = ss.player;
    ss.bullets.push({ x: p.x, y: p.y - p.h / 2 - 6, vx: 0, vy: -8.5, r: 3 });
}

function ssEnemyFire(e) {
    const dx = ss.player.x - e.x;
    const dy = ss.player.y - e.y;
    const dist = Math.sqrt(dx * dx + dy * dy) || 1;
    const spd = 3.1 + Math.min(1.6, ss.frame / 12000);
    ss.enemyBullets.push({
        x: e.x, y: e.y + e.r,
        vx: (dx / dist) * spd,
        vy: (dy / dist) * spd,
        r: 4,
        color: '#ff5b8f',
    });
}

function ssKillEnemy(idx, x, y) {
    const e = ss.enemies[idx];
    ss.score += e.points;
    ssUpdateHUD();
    ssSpawnParticles(e.x, e.y, 16, e.color, 4.2, 30);
    ssSpawnParticles(e.x, e.y, 8, '#ffffff', 2.5, 18);
    ss.rings.push({ x: e.x, y: e.y, r: 4, grow: 2.4, life: 22, color: e.color });
    ss.enemies.splice(idx, 1);
}

function ssHitPlayer(enemyIdx, x, y) {
    ss.lives--;
    ssUpdateHUD();
    ssSpawnParticles(x, y, 22, '#ff2d75', 5, 34);
    ssSpawnParticles(x, y, 12, '#ffffff', 3, 22);
    ss.shake = 14;
    ss.invuln = 100;
    if (enemyIdx >= 0) {
        const e = ss.enemies[enemyIdx];
        ssSpawnParticles(e.x, e.y, 18, e.color, 4.5, 30);
        ss.rings.push({ x: e.x, y: e.y, r: 6, grow: 2.2, life: 26, color: e.color });
        ss.enemies.splice(enemyIdx, 1);
    }
    if (ss.lives <= 0) {
        ss.over = true;
        if (ss.score > ssBest) {
            ssBest = ss.score;
            try { window.localStorage.setItem('shooterBest', String(ssBest)); } catch (e) {}
        }
        ssSpawnParticles(ss.player.x, ss.player.y, 60, '#00f0ff', 6, 50);
        ssSpawnParticles(ss.player.x, ss.player.y, 40, '#ff2d75', 5, 44);
        ss.rings.push({ x: ss.player.x, y: ss.player.y, r: 8, grow: 3, life: 40, color: '#00f0ff' });
    }
}

function ssUpdateHUD() {
    ssScoreEl.textContent = ss.score;
    ssLivesEl.textContent = '❤️'.repeat(Math.max(0, ss.lives));
}

function ssUpdateParticles() {
    for (let i = ss.particles.length - 1; i >= 0; i--) {
        const pt = ss.particles[i];
        pt.x += pt.vx; pt.y += pt.vy; pt.vy += pt.gravity; pt.life--;
        if (pt.life <= 0) ss.particles.splice(i, 1);
    }
    for (let i = ss.rings.length - 1; i >= 0; i--) {
        const rg = ss.rings[i];
        rg.r += rg.grow; rg.life--;
        if (rg.life <= 0) ss.rings.splice(i, 1);
    }
}

function shooterInit() {
    if (shooterRAF) { cancelAnimationFrame(shooterRAF); shooterRAF = null; }
    ss.score = 0; ss.lives = 3;
    ss.over = false; ss.running = true;
    ss.frame = 0; ss.invuln = 0; ss.shake = 0; ss.fireCd = 0; ss.spawnTimer = 55;
    ss.bullets = []; ss.enemyBullets = []; ss.enemies = [];
    ss.particles = []; ss.rings = [];
    ss.player.x = ssW / 2; ss.player.y = 500;
    ss.keys.left = ss.keys.right = ss.keys.up = ss.keys.down = ss.keys.fire = false;
    ss.firing = false; ss.mouse.active = false;
    ssSpawnStars();
    ssUpdateHUD();
    shooterRAF = requestAnimationFrame(shooterLoop);
}

function ssUpdate() {
    ss.frame++;
    for (const st of ss.stars) {
        st.y += st.speed;
        if (st.y > ssH + 2) { st.y = -2; st.x = Math.random() * ssW; }
    }
    if (ss.over) {
        ssUpdateParticles();
        if (ss.shake > 0) ss.shake--;
        return;
    }

    const p = ss.player;

    if (ss.mouse.active) {
        const mx = Math.max(16, Math.min(ssW - 16, ss.mouse.x));
        const my = Math.max(290, Math.min(ssH - 20, ss.mouse.y));
        p.x += (mx - p.x) * 0.18;
        p.y += (my - p.y) * 0.18;
    } else {
        if (ss.keys.left) p.x -= p.speed;
        if (ss.keys.right) p.x += p.speed;
        if (ss.keys.up) p.y -= p.speed;
        if (ss.keys.down) p.y += p.speed;
    }
    p.x = Math.max(16, Math.min(ssW - 16, p.x));
    p.y = Math.max(290, Math.min(ssH - 20, p.y));

    if (ss.fireCd > 0) ss.fireCd--;
    if (ss.keys.fire || ss.firing) ssFire();

    ssUpdateSpawner();

    for (let i = ss.bullets.length - 1; i >= 0; i--) {
        const b = ss.bullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.y < -10) { ss.bullets.splice(i, 1); continue; }
        let hit = false;
        for (let j = ss.enemies.length - 1; j >= 0; j--) {
            const e = ss.enemies[j];
            const dx = b.x - e.x, dy = b.y - e.y;
            if (dx * dx + dy * dy < (e.r + b.r) * (e.r + b.r)) {
                hit = true;
                ssKillEnemy(j, b.x, b.y);
                break;
            }
        }
        if (hit) ss.bullets.splice(i, 1);
    }

    for (let i = ss.enemies.length - 1; i >= 0; i--) {
        const e = ss.enemies[i];
        if (e.type === 'zigzag') {
            e.phase += 0.05;
            e.x += Math.sin(e.phase) * 1.35;
        }
        if (e.type === 'diver') {
            e.dive += 1;
            e.x += Math.sin(e.phase + e.dive * 0.05) * 1.1;
        }
        e.y += e.speed;
        e.fireTimer--;
        if (e.fireTimer <= 0 && e.y > 30 && e.y < ss.player.y + 60) {
            e.fireTimer = ssEnemyFireTimer(e);
            ssEnemyFire(e);
        }
        if (e.y > ssH + 40) { ss.enemies.splice(i, 1); continue; }
        if (ss.invuln <= 0) {
            const dx = e.x - p.x, dy = e.y - p.y;
            if (dx * dx + dy * dy < (e.r + 12) * (e.r + 12)) {
                ssHitPlayer(i, e.x, e.y);
            }
        }
    }

    for (let i = ss.enemyBullets.length - 1; i >= 0; i--) {
        const b = ss.enemyBullets[i];
        b.x += b.vx; b.y += b.vy;
        if (b.y > ssH + 12 || b.y < -12 || b.x < -12 || b.x > ssW + 12) {
            ss.enemyBullets.splice(i, 1); continue;
        }
        if (ss.invuln <= 0) {
            const dx = b.x - p.x, dy = b.y - p.y;
            if (dx * dx + dy * dy < (b.r + 11) * (b.r + 11)) {
                ss.enemyBullets.splice(i, 1);
                ssHitPlayer(-1, b.x, b.y);
            }
        }
    }

    ssUpdateParticles();
    if (ss.invuln > 0) ss.invuln--;
    if (ss.shake > 0) ss.shake--;
}

function ssDrawPlayer(ctx) {
    const p = ss.player;
    if (ss.invuln > 0 && Math.floor(ss.frame / 5) % 2 === 0) return;
    ctx.save();
    ctx.translate(p.x, p.y);
    ctx.fillStyle = '#ff9d2d';
    ctx.shadowColor = '#ff9d2d';
    ctx.shadowBlur = 14;
    ctx.beginPath();
    const fl = 6 + Math.sin(ss.frame * 0.6) * 3;
    ctx.moveTo(-5, 14);
    ctx.lineTo(0, 14 + fl);
    ctx.lineTo(5, 14);
    ctx.closePath();
    ctx.fill();
    ctx.shadowColor = '#00f0ff';
    ctx.shadowBlur = 18;
    const hg = ctx.createLinearGradient(0, -18, 0, 18);
    hg.addColorStop(0, '#eafcff');
    hg.addColorStop(0.5, '#00c3ff');
    hg.addColorStop(1, '#0077ff');
    ctx.fillStyle = hg;
    ctx.beginPath();
    ctx.moveTo(0, -18);
    ctx.lineTo(11, 6);
    ctx.lineTo(6, 10);
    ctx.lineTo(-6, 10);
    ctx.lineTo(-11, 6);
    ctx.closePath();
    ctx.fill();
    ctx.shadowBlur = 8;
    ctx.fillStyle = '#062233';
    ctx.beginPath();
    ctx.ellipse(0, -4, 4.5, 6, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = 'rgba(140, 250, 255, 0.85)';
    ctx.beginPath();
    ctx.ellipse(-1, -6, 2, 3, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowColor = '#00f0ff';
    ctx.shadowBlur = 10;
    ctx.fillStyle = '#00f0ff';
    ctx.beginPath();
    ctx.moveTo(-11, 6);
    ctx.lineTo(-16, 12);
    ctx.lineTo(-8, 10);
    ctx.closePath();
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(11, 6);
    ctx.lineTo(16, 12);
    ctx.lineTo(8, 10);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
}

function ssDrawEnemy(ctx, e) {
    ctx.save();
    ctx.translate(e.x, e.y);
    ctx.shadowColor = e.color;
    ctx.shadowBlur = 14;
    if (e.type === 'drone') {
        ctx.fillStyle = e.color;
        ctx.beginPath();
        ctx.moveTo(0, -e.r);
        ctx.lineTo(e.r, 0);
        ctx.lineTo(0, e.r);
        ctx.lineTo(-e.r, 0);
        ctx.closePath();
        ctx.fill();
        ctx.shadowBlur = 6;
        ctx.fillStyle = 'rgba(255,255,255,0.75)';
        ctx.beginPath();
        ctx.arc(0, 0, 3.5, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = 'rgba(4,6,15,0.6)';
        ctx.beginPath();
        ctx.arc(0, 0, 1.6, 0, Math.PI * 2);
        ctx.fill();
    } else if (e.type === 'zigzag') {
        ctx.strokeStyle = e.color;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(0, -e.r - 2);
        ctx.lineTo(e.r - 4, -2);
        ctx.lineTo(2, -2);
        ctx.lineTo(e.r - 2, e.r - 2);
        ctx.lineTo(-2, e.r - 2);
        ctx.lineTo(-e.r + 4, -2);
        ctx.lineTo(-2, -2);
        ctx.closePath();
        ctx.stroke();
        ctx.fillStyle = e.color;
        ctx.beginPath();
        ctx.arc(0, 0, 4, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = 'rgba(4,6,15,0.6)';
        ctx.beginPath();
        ctx.arc(0, 0, 1.8, 0, Math.PI * 2);
        ctx.fill();
    } else {
        ctx.fillStyle = e.color;
        ctx.beginPath();
        ctx.moveTo(0, -e.r - 2);
        ctx.lineTo(e.r * 0.8, e.r * 0.5);
        ctx.lineTo(0, e.r * 0.2);
        ctx.lineTo(-e.r * 0.8, e.r * 0.5);
        ctx.closePath();
        ctx.fill();
        ctx.shadowBlur = 6;
        ctx.fillStyle = '#fff3d6';
        ctx.beginPath();
        ctx.arc(0, -3, 3, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.restore();
}

function ssDraw() {
    const ctx = ssCtx;
    ctx.clearRect(0, 0, ssW, ssH);
    ctx.save();
    if (ss.shake > 0) {
        ctx.translate((Math.random() - 0.5) * ss.shake * 0.7, (Math.random() - 0.5) * ss.shake * 0.7);
    }

    const bg = ctx.createLinearGradient(0, 0, 0, ssH);
    bg.addColorStop(0, '#0a0f22');
    bg.addColorStop(0.5, '#060a18');
    bg.addColorStop(1, '#04060f');
    ctx.fillStyle = bg;
    ctx.fillRect(-20, -20, ssW + 40, ssH + 40);

    const ng = ctx.createRadialGradient(ssW * 0.8, ssH * 0.25, 20, ssW * 0.8, ssH * 0.25, 260);
    ng.addColorStop(0, 'rgba(157, 77, 255, 0.10)');
    ng.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = ng;
    ctx.fillRect(0, 0, ssW, ssH);
    const ng2 = ctx.createRadialGradient(ssW * 0.15, ssH * 0.7, 20, ssW * 0.15, ssH * 0.7, 240);
    ng2.addColorStop(0, 'rgba(0, 240, 255, 0.08)');
    ng2.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = ng2;
    ctx.fillRect(0, 0, ssW, ssH);

    for (const st of ss.stars) {
        const a = 0.25 + st.layer * 0.3;
        ctx.globalAlpha = a * (0.7 + 0.3 * Math.sin(st.tw + ss.frame * 0.06));
        ctx.fillStyle = st.layer === 2 ? '#cfe9ff' : st.layer === 1 ? '#9ad0ff' : '#5a7db8';
        ctx.beginPath();
        ctx.arc(st.x, st.y, st.size, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;

    ctx.textAlign = 'left';
    ctx.textBaseline = 'top';
    ctx.font = 'bold 16px sans-serif';
    ctx.shadowColor = '#00f0ff';
    ctx.shadowBlur = 14;
    ctx.fillStyle = '#eafcff';
    ctx.fillText('🚀 Neon Shooter', 14, 14);
    ctx.shadowBlur = 0;

    for (const b of ss.bullets) {
        ctx.save();
        ctx.shadowColor = '#00f0ff';
        ctx.shadowBlur = 12;
        ctx.fillStyle = '#a6fbff';
        ctx.beginPath();
        ctx.roundRect(b.x - 2, b.y - 9, 4, 14, 2);
        ctx.fill();
        ctx.restore();
    }

    for (const e of ss.enemies) ssDrawEnemy(ctx, e);

    for (const b of ss.enemyBullets) {
        ctx.save();
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 12;
        ctx.fillStyle = '#ffe3ec';
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = b.color;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.r * 0.55, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }

    if (!ss.over) ssDrawPlayer(ctx);

    for (const rg of ss.rings) {
        const a = Math.max(0, rg.life / 30);
        ctx.globalAlpha = a;
        ctx.strokeStyle = rg.color;
        ctx.lineWidth = 2.5;
        ctx.shadowColor = rg.color;
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.arc(rg.x, rg.y, rg.r, 0, Math.PI * 2);
        ctx.stroke();
    }
    ctx.globalAlpha = 1;
    ctx.shadowBlur = 0;

    for (const pt of ss.particles) {
        const a = Math.max(0, pt.life / pt.maxLife);
        ctx.globalAlpha = a;
        ctx.fillStyle = pt.color;
        ctx.shadowColor = pt.color;
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, pt.size * a, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalAlpha = 1;
    ctx.shadowBlur = 0;

    ctx.restore();

    if (ss.over) {
        ssDrawGameOver(ctx);
        return;
    }

    if (ss.frame < 300) {
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.font = 'bold 13px sans-serif';
        ctx.shadowColor = '#00f0ff';
        ctx.shadowBlur = 8;
        ctx.fillStyle = 'rgba(191, 238, 252, 0.9)';
        ctx.fillText('← → move  •  ↑ ↓ fine-aim  •  SPACE fire', ssW / 2, ssH - 22);
        ctx.shadowBlur = 0;
    }
}

function shooterLoop() {
    shooterRAF = requestAnimationFrame(shooterLoop);
    if (!ss.running) return;
    ssUpdate();
    ssDraw();
}

document.addEventListener('keydown', (e) => {
    if (currentScreen !== 'shooter') return;
    if (e.key === 'ArrowLeft' || e.key === 'ArrowRight' || e.key === 'ArrowUp' || e.key === 'ArrowDown') {
        e.preventDefault();
        ss.mouse.active = false;
        if (e.key === 'ArrowLeft') ss.keys.left = true;
        else if (e.key === 'ArrowRight') ss.keys.right = true;
        else if (e.key === 'ArrowUp') ss.keys.up = true;
        else ss.keys.down = true;
    } else if (e.key === ' ' || e.key === 'Spacebar') {
        e.preventDefault();
        ss.keys.fire = true;
    }
});

document.addEventListener('keyup', (e) => {
    if (currentScreen !== 'shooter') return;
    if (e.key === 'ArrowLeft') ss.keys.left = false;
    else if (e.key === 'ArrowRight') ss.keys.right = false;
    else if (e.key === 'ArrowUp') ss.keys.up = false;
    else if (e.key === 'ArrowDown') ss.keys.down = false;
    else if (e.key === ' ' || e.key === 'Spacebar') ss.keys.fire = false;
});

    ss.mouse.active = true;
    ss.mouse.x = (e.clientX - rect.left) * (ssW / rect.width);
    ss.mouse.y = (e.clientY - rect.top) * (ssH / rect.height);
});
